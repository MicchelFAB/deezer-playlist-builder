┌─────────────────────────────────────────────┐
│  📦 INSTALAÇÃO (Primeira vez)               │
└─────────────────────────────────────────────┘

  ./setup.sh


┌─────────────────────────────────────────────┐
│  🔑 OBTER CREDENCIAIS DO NAVEGADOR          │
└─────────────────────────────────────────────┘

  1. Abra: https://www.deezer.com
  2. F12 → Network → Adicione 1 música
  3. Encontre: gw-light.php?method=playlist.addSongs
  4. Copy as cURL
  5. Extraia:
     • api_token=XXXX → DEEZER_API_TOKEN
     • x-deezer-user: 123 → DEEZER_USER_ID
     • Cookie: ... → DEEZER_COOKIES (tudo!)


┌─────────────────────────────────────────────┐
│  ⚙️  CONFIGURAR                              │
└─────────────────────────────────────────────┘

  nano .env.local

  # Cole os valores:
  export DEEZER_API_TOKEN="..."
  export DEEZER_USER_ID="..."
  export DEEZER_COOKIES="sid=...; arl=...; jwt=..."
  export DEEZER_PLAYLIST_ID="..."


┌─────────────────────────────────────────────┐
│  🔍 BUSCAR MÚSICAS                          │
└─────────────────────────────────────────────┘

  # Crie artist.txt com um artista por linha
  echo "Limão Com Mel" > artist.txt
  echo "Capim Com Mel" >> artist.txt

  # Execute busca
  node search-albums.js \
    --inputFile artist.txt \
    --outputFile playlist_tracks.txt


┌─────────────────────────────────────────────┐
│  ➕ ADICIONAR À PLAYLIST                    │
└─────────────────────────────────────────────┘

  # Carregue credenciais
  source .env.local

  # Opção 1: Bash (rápido)
  ./add-tracks-manual.sh

  # Opção 2: TypeScript (recomendado)
  ts-node add-tracks.ts --inputFile playlist_tracks.txt
